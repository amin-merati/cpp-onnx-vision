cmake_minimum_required(VERSION 3.18)
project(cpp_onnx_vision VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ONNXRUNTIME_ROOT "" CACHE PATH "Extracted ONNX Runtime C/C++ SDK")
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ORT_LIBRARY onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
add_library(vision_core src/image.cpp)
target_include_directories(vision_core PUBLIC include)
target_compile_options(vision_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(vision_infer src/main.cpp src/classifier.cpp)
target_include_directories(vision_infer SYSTEM PRIVATE "${ORT_INCLUDE}")
target_link_libraries(vision_infer PRIVATE vision_core "${ORT_LIBRARY}")
target_compile_options(vision_infer PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE vision_core)
  add_test(NAME core_tests COMMAND core_tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME reference_parity COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/reference_test.py"
    --exe $<TARGET_FILE:vision_infer> --work "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
endif()
if(ENABLE_SANITIZERS)
  foreach(target vision_core vision_infer core_tests)
    if(TARGET ${target})
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
endif()
